// LKS de tanýmlanmýþ en son hesap planýný bulur
// HastaIlkHarfKullan seçeneðinde bulunan deðere göre sorgulama deðiþir.
function GetLastCustomerAccountCode: Integer;
var
  sql: string;
  ilkharf  : string;
begin
  ilkharf := Copy(FaturaListesi.FieldByName('ADSOYAD').AsString, 1, 1);
  if (HastaIlkHarfKullan) then 
    sql := 'SELECT TOP 1 CODE,SUBSTRING(CODE,LEN('''+ HesapPlaniOnEki + ilkharf +''') + 1,500) FROM LG_' + firmaNo + '_EMUHACC WHERE CODE LIKE '''+ HesapPlaniOnEki + ilkharf +'%'' ORDER BY CODE DESC'
  else
    sql := 'SELECT TOP 1 CODE,SUBSTRING(CODE,LEN('''+ HesapPlaniOnEki +''') + 1,500) FROM LG_' + firmaNo + '_EMUHACC ORDER BY CODE DESC';
  with query_exec(LKSConnection,sql,[],[]) do 
  try
    Open;
    if (RecordCount > 0) then 
      Result := StrToInt(Fields.Fields[1].AsString)
    else       
      if HastaIlkHarfKullan then 
        Result := 1
      else
        Result := 1;
  finally
    Free;
  end;
end;

// önek+dosyano döndürür

end;

// LKS de müþteri var mý yok mu kontrol eder.
function CustomerExists(ACustomerCode: string): Boolean;
begin  
  with query_exec(LKSConnection,'SELECT CODE FROM LG_'+ firmaNo +'_CLCARD WHERE (ACTIVE=0) AND (CARDTYPE=3) AND CODE LIKE '''+
    ACustomerCode +'%''',[],[]) do 
  try
    Open;
    Result := RecordCount > 0;
  finally
    Free;
  end;  
end;

// En son olabilecek hesap kodunu döndürür.
function GetCustomerAccountCode: string;
var
  ilkHarf : string;
begin
  if HastaIlkHarfKullan then begin
    ilkharf := Copy(faturaBaslik.FieldByName('AD').AsString, 1, 1);
    if hesapCache.IndexOfName(ilkHarf) = -1 then 
      hesapCache.Values[ilkHarf] := IntToStr(GetLastCustomerAccountCode);
    Result := HesapPlaniOnEki + ilkHarf + LeadingZero(StrToInt(hesapCache.Values[ilkHarf]) + 1,HesapPlaniSifirSayisi);
  end else begin
    lastCode := lastCode + 1;
    Result := HesapPlaniOnEki + LeadingZero(lastCode + 1,HesapPlaniSifirSayisi)    
  end;    
end;

begin

// fatura ile ilgili toplamlarý hesaplar.  
procedure ToplamlariHesapla;
var
  satirToplam : Double;
  iskonto     : Double; 
  satirKDV    : Double;
begin
  _FaturaBrutToplami    := 0;
  _FaturaKDVMatrahi     := 0;
  _FaturaIskontoToplami := 0;
  _FaturaNetToplam      := 0;
  _FaturaKDVsi          := 0;
  while not faturaDetay.Eof do
    begin
      iskonto := 0;
      // Satýr ham tutarýný bul
      satirToplam := faturaDetay.FieldByName('BIRIMFIYAT').AsFloat *
        faturaDetay.FieldByName('ADET').AsFloat;
      // KDV dahil mi ?
      if (faturaBaslik.FieldByName('KDVDURUM').AsString = 'Dahil') then
        // öyleyse kdv hariç toplamý hesapla
        satirToplam := satirToplam / (1 + (faturaDetay.FieldByName('KDV').AsFloat / 100));
      // fatura brut toplamýna ekle
      _FaturaBrutToplami := _FaturaBrutToplami + satirToplam;
      // iskonto tanýmlanmýþ mý ?
      if (faturaDetay.FieldByName('ISKONTO').AsFloat > 0) then
        // öyleyse satýr iskontosunu hesapla
        iskonto := (satirToplam * faturaDetay.FieldByName('ISKONTO').AsFloat) / 100;
      // kdv matrahýna ekle
      _FaturaKDVMatrahi := _FaturaKDVMatrahi + ( satirToplam - iskonto );
      // satýr iskontosunu fatura iskontosuna ekle
      _FaturaIskontoToplami := _FaturaIskontoToplami + iskonto;
      // satýr kdvsini hesapla
      satirKDV := ( ( satirToplam - iskonto ) * faturaDetay.FieldByName('KDV').AsFloat ) / 100;
      // fatura kdvsine ekle  
      _FaturaKDVsi := _FaturaKDVsi + satirKDV;      
      faturaDetay.Next;
    end;
  // net toplamý hesapla
  _FaturaNetToplam := _FaturaKDVMatrahi + _FaturaKDVsi;     
end;

procedure TransformDiscountToTransaction(ADiscountRate,ADiscountAmount: Double;Root: TXMLItem);
begin
  with Root.New do begin
    Name := 'TRANSACTION';
    with New do begin Name := 'TYPE'; Text := '2'; end;
    with New do begin Name := 'QUANTITY';Text := '0';end;
    with New do begin Name := 'RC_XRATE';Text := FloatToStr(ADiscountAmount);end;
    with New do begin Name := 'DISCOUNT_RATE';Text := FloatToStr(ADiscountRate);end;
    with New do begin Name := 'UNIT_CODE';Text := '';end;
    with New do begin Name := 'UNIT_CONV1';Text := '0';end;
    with New do begin Name := 'UNIT_CONV2';Text := '0';end;
    with New do begin Name := 'BILLED';Text := '1';end;    
  end;
end;

// Fatura kalemleri için xml oluþturur.
procedure TransformInvoiceDetails(Root: TXMLItem);
var
  det             : TXMLItem;
  itm             : TXMLItem;
  discountAmount  : Double;
  node            : TXMLItem;
  kdvNode         : TXMLItem;
  unitPrice       : Double;
  vatBase         : Double;
  vatAmount       : Double;
  totalNet        : Double;
  vatRate         : Double;
  lineTotal       : Double;
  tmp             : Double;
  stockCode       : string;
  stockAccCode    : string;
begin
  if not faturaKalemleriDonusmesin then begin
    if (Trim(faturaDetay.FieldByName('KOD').AsString) = '') then begin
      MakeError('Fatura kaleminin iþlem kodu boþ!');
      Exit;
    end;
    // hizmet eþlemesi bulunuyor....
    stockCode := faturaDetay.FieldByName('KOD').AsString; // ana deðiþken burada geçici deðiþkenmiþ gibi kullanýlýyor
    if (Copy(stockCode,Length(stockCode),1) = 'K') then // son harfinde K var mý ? (yani katký mý ?)
      // evet
      stockCode := Copy(stockCode, 1, Length(stockCode) - 1); // o zaman sonundaki K yý çýkar
      
    node := GetNode('LKS/ESLEMELER/HIZMET_ESLEMELERI/HIZMET_ESLEMESI?kod=' + 
      stockCode,configuration.Root); // eþleme bilgisine bak
    // Kontroller yapýlýyor...
    if (Trim(node.Params.Values['donusmus1']) = '') then begin
      MakeError('Hizmet eþlemesinde karþý kod alaný boþ ! : ' + faturaDetay.FieldByName('KOD').AsString);
      Exit;    
    end;
    if (Trim(node.Params.Values['donusmus2']) = '') then begin
      MakeError('Hizmet eþlemesinde karþý muhasebe kodu alaný boþ ! : ' + faturaDetay.FieldByName('KOD').AsString);
      Exit;    
    end;
    // program buraya gelebiliyorsa þu ana kadar herhangi problem olmamýþ
    // öylese dönüþmüþ kodlarý ana deðiþkenlere at.
    stockCode := node.Params.Values['donusmus1']; 
    stockAccCode := node.Params.Values['donusmus2'];    
  end else begin
    // eðer sabit kodlar kullanýlmasý iþaretlenmiþse
    // o zaman parameterelerden gelen deðerleri ata
    stockCode := sabitFaturaKalemiKod;
    stockAccCode := sabitFaturaKalemiMuhKod;
    // kontrol et boþ olarak gelmiþ mi?
    if (Trim(stockCode) = '') then begin
      MakeError('Hizmet eþlemesi için sabit kod atamasý yapýlmamýþ!');
      Exit;
    end;
    if (Trim(stockAccCode) = '') then begin
      MakeError('Hizmet eþlemesi için sabit muhasebe kod atamasý yapýlmamýþ!');
      Exit;
    end;
  end;
  // kdv eþlemesi bulunuyor...
  kdvNode := GetNode('LKS/ESLEMELER/KDV_ESLEMELERI/KDV_ESLEMESI?kod=' + 
    faturaDetay.FieldByName('KDV').AsString,configuration.Root);
  // kontroller yapýlýyor...
  if (not Assigned(kdvNode)) then begin
    MakeError('KDV Eþlemesi bulunamadý! : ' + faturaDetay.FieldByName('KDV').AsString);
    Exit;    
  end;
  if (Trim(kdvNode.Params.Values['donusmus1']) = '') then begin
    MakeError('KDV eþlemesinde karþý muhasebe kodu boþ ! : ' + faturaDetay.FieldByName('KDV').AsString);
    Exit;    
  end;  
  if (faturaDetay.FieldByName('ISKONTO').AsFloat < 0) then begin
    MakeError('Eksi iskonto içeren faturalar aktarýlamaz! : ' + faturaDetay.FieldByName('ACIKLAMA').AsString);
    Exit;    
  end;
  // bütün hesaplamalar kdv hariç olarak hesaplanacak
  //----------------------------------------------------------------- 
  // birim fiyat hesaplamasý
  // bu hesaba iskonto dahil deðildir.
  if (faturaBaslik.FieldByName('KDVDURUM').AsString = 'Dahil') then
    // BIRIMFIYAT kdv dahil olarak hesaplanmýþtýr.Öyleyse KDV yi ayýr. birim_fiyat = BIRIMFIYAT - ( ( BIRIMFIYAT * KDV ) / (100 + KDV)
    unitPrice := faturaDetay.FieldByName('BIRIMFIYAT').AsFloat -
	  ((faturaDetay.FieldByName('BIRIMFIYAT').AsFloat * 
	    faturaDetay.FieldByName('KDV').AsFloat) / (100 +
		faturaDetay.FieldByName('KDV').AsFloat))
  else 
    unitPrice := faturaDetay.FieldByName('BIRIMFIYAT').AsFloat; // birim_fiyat = BIRIMFIYAT kdv hariçtir.
  // satýr brüt toplamý
  lineTotal := unitPrice * faturaDetay.FieldByName('ADET').AsInteger; // satýr_toplam = birim_fiyat * adet
  // satýr toplam iskontosu iskonto hesaplamasý  
  discountAmount := (lineTotal * faturaDetay.FieldByName('ISKONTO').AsFloat) / 100; // yapýlan_iskonto = ( satýr_toplam * ISKONTO )  / 100
  // kdv matrahý hesaplanýyor satýr brüt toplamý - iskonto miktarý
  vatBase := lineTotal - discountAmount; // kdv_matrahý = satýr_toplam - yapýlan_iskonto
  // kdv'nin hesaplanmasý 
  vatRate := faturaDetay.FieldByName('KDV').AsFloat; // kdv oraný
  if (faturaBaslik.FieldByName('KDVDURUM').AsString <> 'Muaf') then // muaf deðilse
    vatAmount := (vatBase * vatRate) / 100 // hesaplanan_kdv = kdv_matrahý * kdv_oraný
  else
    vatAmount := 0; // hesaplanan_kdv = 0  
  // toplam net tutar hesaplanýyor ( kdv yi de içerir)
  totalNet := vatBase + vatAmount; // satýr_toplam = kdv_matrahý + hesaplanan_kdv
  // *********************************************
  // xml oluþturuluyor...
  det := Root.New;det.Name := 'TRANSACTION';
  itm := det.New;itm.Name := 'TYPE';
  itm.Text := faturaKalemTipi; // 0 - Malzeme 4 - Hizmet
  itm := det.New;itm.Name := 'MASTER_CODE';
  itm.Text := stockCode;
  itm := det.New;itm.Name := 'GL_CODE1';
  itm.Text := stockAccCode;
  itm := det.New;itm.Name := 'GL_CODE2';
  itm.Text := kdvNode.Params.Values['donusmus1'];
  itm := det.New;itm.Name := 'QUANTITY';
  itm.Text := faturaDetay.FieldByName('ADET').AsString;
  if (faturaTipi = '8') or (faturaTipi = '9') then begin   // toptan ise kdv haric
    itm := det.New;itm.Name := 'PRICE';  
    itm.Text := FloatToStr(unitPrice); 
    itm := det.New;itm.Name := 'TOTAL';
    itm.Text := FloatToStr(unitPrice * faturaDetay.FieldByName('ADET').AsInteger);
  end else if (faturaTipi = '7') then begin // perakende fatura için kdv dahil 
    tmp := unitPrice * (1 + (vatRate / 100)); 
    itm := det.New;itm.Name := 'PRICE';  
    itm.Text := FloatToStr(tmp); 
    itm := det.New;itm.Name := 'TOTAL';
    itm.Text := FloatToStr(tmp * faturaDetay.FieldByName('ADET').AsInteger);     
  end;
  itm := det.New;itm.Name := 'RC_XRATE';
  itm.Text := '1';
  if discountAmount > 0 then begin
    itm := det.New;itm.Name := 'COST_DISTR';  
    itm.Text := FloatToStr(discountAmount);// iskonto tutarý
    itm := det.New;itm.Name := 'DISCOUNT_DISTR'; 
    itm.Text := FloatToStr(discountAmount); 
  end;
  itm := det.New;itm.Name := 'UNIT_CODE';
  itm.Text := 'ADET';
  itm := det.New;itm.Name := 'UNIT_CONV1';
  itm.Text := '1';
  itm := det.New;itm.Name := 'UNIT_CONV2';
  itm.Text := '1';
   
  itm := det.New;itm.Name := 'VAT_INCLUDED';
  if (faturaTipi = '8') or (faturaTipi = '9') then
    itm.Text := '0'
  else if (faturaTipi = '7') then 
    itm.Text := '1';
  itm := det.New;itm.Name := 'VAT_RATE';
  itm.Text := faturaDetay.FieldByName('KDV').AsString;  // kdv oraný
  itm := det.New;itm.Name := 'VAT_AMOUNT';	  
  itm.Text := FloatToStr(vatAmount);// hesaplanan_kdv
  itm := det.New;itm.Name := 'VAT_BASE';
  itm.Text := FloatToStr(vatBase);// kdv matrahý  
  itm := det.New;itm.Name := 'BILLED';
  itm.Text := '1';	
  itm := det.New;itm.Name := 'TOTAL_NET';
  itm.Text := FloatToStr(vatBase);	// satýr toplam kdv hariç
  {itm := det.New;itm.Name := 'DETAILS';
  itm := det.New;itm.Name := 'EDT_CURR';
  itm.Text := '160';
  itm := det.New;itm.Name := 'EDT_PRICE';
  itm.Text := FloatToStr(unitPrice);
  itm := det.New;itm.Name := 'GENIUSFLDSLIST';}
  // eðer satýr a iskonto uygulanmýþsa ayrýyetten bunu belirten bir iskonto hareketinin eklenmesi gerekiyor
  if (discountAmount > 0) then 
    TransformDiscountToTransaction(faturaDetay.FieldByName('ISKONTO').AsFloat,discountAmount,Root);
end;

function GetInstitueNode: TXMLItem;
var
  node : TXMLItem;
begin
  Result := nil;
  node := GetNode('LKS/AYARLAR/GENEL_AYARLAR/AYAR?adi=referansli_kurumlar/REFERANSLAR/REFERANS?adi=' + 
    faturaBaslik.FieldByName('KURUM').AsString,configuration.Root);
  if (Trim(faturaBaslik.FieldByName('KURUM').AsString) = '') then begin
    MakeError('Kurum alaný boþ!');
    Exit;
  end;
  if node.Params.Values['kullan'] = '1' then begin
    // bu kurum referans kullanýyor o zaman
    // eþleme bilgisi için referanslarý kullanacaðýz.
    if (Trim(faturaBaslik.FieldByName('REFERANSKOD').AsString) = '') then begin
      MakeError(faturaBaslik.FieldByName('KURUM').AsString +' kurumu için referans kullanýlacak fakat Referans Belirtilmemiþ!');
      Exit;
    end;

      faturaBaslik.FieldByName('REFERANSKOD').AsString,configuration.Root);    
  end else begin

    // öyleyse kurum eþlemelerine bakacaðýz
    result := GetNode('LKS/ESLEMELER/KURUM_ESLEMELERI/KURUM_ESLEMESI?kod=' + 
      faturaBaslik.FieldByName('KURUM').AsString,configuration.Root);    
  end;  
end;

// Kasadan kesilmiþ nakit fatura xml i oluþturur. Araçlar -> Veri aktarýmý (içeri) -> Kasa - Kasa Ýþlemleri kýsmýndan alýnýr.
procedure TransformCashInvoice;
var
  node : TXMLItem;
begin
  if (nakitKasaKodu = '') then begin
    MakeError('Nakit kasa kodu atanmamýþ!');
    Exit;
  end;
  if (nakitKasaMuhasebeKodu = '') then begin
    MakeError('Nakit kasa muhasebe kodu boþ!');
    Exit;
  end;
  baslikXml.Root.Name := 'SD_TRANSACTIONS';
  with baslikXml.Root.New do begin
    Name := 'SD_TRANSACTION';
    Params.Add('DBOP=INS');
    with New do begin Name := 'TYPE'; Text := kasaIslemTipi; end;
    with New do begin Name := 'SD_CODE'; Text := nakitKasaKodu; end;
    with New do begin Name := 'SD_CODE_CROSS';end;
    with New do begin Name := 'SD_NUMBER_CROSS';end;
    with New do begin Name := 'CROSS_DATA_REFERENCE';end;
    with New do begin Name := 'GL_CODE1'; Text := musteriMuhasebeKodu; end;// cari kodu
    with New do begin Name := 'GL_CODE2'; Text := nakitKasaMuhasebeKodu; end;// kasa muhasebe kodu
    with New do begin Name := 'DATE'; Text := FormatDateTime('dd/MM/yyyy',faturaBaslik.FieldByName('FATURATARIH').AsDateTime); end;
    //with New do begin Name := 'NUMBER'; Text := faturaBaslik.FieldByName('FATURANO').AsString; end;
    with New do begin Name := 'MASTER_TITLE'; Text := faturaBaslik.FieldByName('AD').AsString + ' ' + 
      faturaBaslik.FieldByName('SOYAD').AsString; end;
    with New do begin Name := 'AMOUNT'; Text := FloatToStr(_FaturaNetToplam); end;
    with New do begin Name := 'RC_XRATE'; Text := '1'; end;
    with New do begin Name := 'RC_AMOUNT'; Text := FloatToStr(_FaturaNetToplam); end;
    with New do begin Name := 'TC_AMOUNT'; Text := FloatToStr(_FaturaNetToplam); end;
    with New do begin Name := 'CREATED_BY'; Text := '1'; end;
    with New do begin Name := 'DATE_CREATED'; Text := FormatDateTime('dd/MM/yyyy',faturaBaslik.FieldByName('FATURATARIH').AsDateTime); end;
    with New do begin Name := 'CURRSEL_TOTALS'; Text := '2'; end;
    node := New;
    node.Name := 'ATTACHMENT_INVOICE';
    BaseInvoice(node);    
  end;  
end;

// fatura üst bilgilerini kontrol eder.
function CheckInvoice: Boolean;
begin
  Result := True;
  if (Trim(faturaBaslik.FieldByName('FATURANO').AsString) = '') then begin
    MakeError('Fatura Numarasý Boþ');
    Result := False;
    Exit;
  end;
  if (faturaBaslik.FieldByName('FATURA_TUTARI').AsFloat = 0) then begin
    MakeError('Fatura Tutarý 0 ');
    Result := False;
    Exit;
  end;
  if (faturaDetay.RecordCount = 0) then begin
    MakeError('Fatura detaylarý yok!');
    Result := False;
    Exit;
  end;
  if (faturaBaslik.FieldByName('KURUM').AsString = '') then begin
    MakeError('Kurum bilgisi boþ!');
    Result := False;
    Exit;
  end;
end;

// fatura xml leri oluþturulmadan önce gereken parametrelerin atanmasýný saðlar.
procedure BeforeTransform;
var
  node  : TXMLItem;
begin
  if (faturalamaModeli = FATURAMODEL_GELISLER_KURUM) then begin
    node := GetInstitueNode;
    if (not Assigned(node)) then Exit;
    if (Trim(node.Params.Values['donusmus1']) = '') then begin
      if node.Name = 'REFERANS_ESLEMESI' then
        MakeError('Referans eþlemesinde karþý kod alaný boþ ! : ' + faturaBaslik.FieldByName('REFERANS').AsString)
      else
        MakeError('Kurum eþlemesinde karþý kod alaný boþ ! : ' + faturaBaslik.FieldByName('KURUM').AsString);
      Exit;    
    end;
    if (Trim(node.Params.Values['donusmus2']) = '') then begin
      if node.Name = 'REFERANS_ESLEMESI' then
        MakeError('Referans eþlemesinde karþý muhasebe kodu alaný boþ ! : ' + faturaBaslik.FieldByName('REFERANS').AsString)
      else
        MakeError('Kurum eþlemesinde karþý muhasebe kodu alaný boþ ! : ' + faturaBaslik.FieldByName('KURUM').AsString);
      Exit;    
    end;  
    musteriKodu := node.Params.Values['donusmus1'];
    musteriMuhasebeKodu := node.Params.Values['donusmus2'];
  end else if (faturalamaModeli = FATURAMODEL_HASTA_KAYDI) then begin
    if hastaIcinKayitOlustur then begin    
      if CustomerExists(GetCustomerCode) then begin
        with query_exec(LKSConnection,StringReplace(musteriSorgusu,'%firmcode%',firmaNo),['%musteri_kodu%'],[GetCustomercode]) do
        try
          Open;
          if (RecordCount > 0) then begin
            musteriKodu := Fields.Fields[0].AsString;
            musteriMuhasebeKodu := Fields.Fields[1].AsString;
          end else begin
            MakeError('Müþteri kaydý bulunamadý! : ' + GetCustomerCode);
            Exit;
          end;          
        finally
          Free;      
        end;// try
      end else begin// if
        if preventDuplicatesList.IndexOf(faturaListesi.FieldByName('DOSYANO').AsString) = -1 then begin
          BuildCustomerXml;
          preventDuplicatesList.Add(faturaListesi.FieldByName('DOSYANO').AsString);
        end;
      end; // else
    end else begin // if hastaIcinKayitOlustur
      musteriKodu := faturaBaslik.FieldByName('CARIKODU').AsString;
      musteriMuhasebeKodu := faturaBaslik.FieldByName('MUHASEBEKODU').AsString;
      if (Trim(musteriKodu) = '') then begin
        MakeError('Hasta / LKS Müþteri eþleþtirilmesi yapýlmamýþ. Karþý Kodu alaný boþ.');
        Exit;
      end;
      if (Trim(musteriMuhasebeKodu) = '') then begin
        MakeError('Hasta / LKS Müþteri eþleþtirilmesi yapýlmamýþ. Karþý Muhasebe Kodu alaný boþ.');
        Exit;
      end;
    end;    
  end;// else
  
end;

// dönüþümün baþlangýç yeri
// procedure Main bu procedure ü çalýþtýrmak zorunda yoksa 
// hiç birþey oluþturulmaz.
procedure TransformInvoice;
begin      
  if exportPath = '' then begin
    MakeError('Aktarým yolu tanýmlanmamýþ!');
    Exit;
  end;  
  faturaBaslik := query_exec(SystemConnection,
    'SELECT K.DOSYANO,FB.FATURATARIH,FB.FATURANO,FB.KDVDURUM,FB.TURU,FB.FATURA_TUTARI,K.CARIKODU,K.MUHASEBEKODU,' +
    'FB.ACIKLAMA,G.KURUM,G.REFERANSKOD,G.REFERANS,K.AD,K.SOYAD,K.SICILNO,K.ADRES,K.ILCE,K.IL,K.VATANDASLIKNO,' +
    'K.KARNENO FROM FATBASLIK FB LEFT OUTER JOIN KIMLIK K ON FB.DOSYANO = K.DOSYANO ' +
    'LEFT OUTER JOIN GELISLER G ON (FB.DOSYANO = G.DOSYANO ) AND (FB.GELISNO = G.GELISNO) ' +
    'LEFT OUTER JOIN REHBER R ON G.REFERANSKOD = R.KOD ' +
    'LEFT OUTER JOIN KURUM KUR ON G.KURUM = KUR.KURUM   ' +
    ' WHERE (FB.DOSYANO=!dosyano) and (FB.GELISNO=!gelisno) and (FB.KARTNO = !kartno)',
    ['!dosyano','!gelisno','!kartno'],[DosyaNo,GelisNo,KartNo]);
  faturaDetay := query_exec(SystemConnection,
    'SELECT SIRANO, SEC, TARIH, TUR, KOD, ACIKLAMA,ADET, BIRIM, MIKTAR, ' +
  	'BIRIMFIYAT, TUTAR, ISKONTO, KDV, BUTCEKODU, KASA, ONAY, KULLANICI '+
  	' FROM FATURA F ' +
    ' WHERE (DOSYANO=!dosyano) and (GELISNO=!gelisno) and (KARTNO = !kartno) ' +
  	' ORDER BY SIRANO',
    ['!dosyano','!gelisno','!kartno'],[DosyaNo,GelisNo,KartNo]);
   
  baslikXml := TECXMLParser.Create(nil);  
  try
    faturaBaslik.Open;
    faturaDetay.Open;
    BeforeTransform;
    if (not SUCCESS) then Exit;
    if CheckInvoice then begin 
      ToplamlariHesapla;
      if (Abs(_FaturaNetToplam - faturaBaslik.FieldByName('FATURA_TUTARI').AsFloat) > 3) then begin
        MakeError('Fatura toplamý ile hesaplanan toplam arasýnda fark var! HT: ' + 
          FloatToStr(_FaturaNetToplam) + ' FT: ' + FloatToStr(faturaBaslik.FieldByName('FATURA_TUTARI').AsFloat) +
          ' Bu sorun genellike eksi (-) iskontolardan kaynaklanmýþ olabilir.');
        Exit;
      end;
      if (faturaBaslik.FieldByName('TURU').AsString = 'Açýk') then      
        TransformForwardedInvoice
      else
        TransformCashInvoice;
      if SUCCESS then
        if faturaBaslik.FieldByName('TURU').AsString = 'Açýk' then
          baslikXml.SaveToFile(exportPath + GetXmlFileName('1739'))
        else
          baslikXml.SaveToFile(exportPath + GetXmlFileName('1854'));
    end;
  finally    
    baslikXml.Free;
    faturaBaslik.Free;
    faturaDetay.Free;
  end;
end;

  exportPath := node.Params.Values['yol'];
  node := GetNode('LKS/AYARLAR/GENEL_AYARLAR/AYAR?adi=logo_ozel_kod',configuration.Root);
  logoOzelKod := node.Params.Values['ozelkod'];
  {
  faturaKalemleriDonusmesin : Boolean;
