END

CREATE TRIGGER [dbo].[TG_StokFiyatEkle] ON [dbo].[STOKLAR] 
FOR INSERT, DELETE AS BEGIN SET NOCOUNT ON; DECLARE @Kur VARCHAR(5); 
DECLARE @YeniStoklar TABLE (ID INT NOT NULL PRIMARY KEY, ANABIRIM INT NULL); 
SELECT @Kur = ANAHTAR FROM dbo.GENINI WHERE BOLUM = -10135; 
IF EXISTS (SELECT 1 FROM deleted) AND EXISTS (SELECT 1 FROM inserted) RETURN; 
IF EXISTS (SELECT 1 FROM inserted) 
BEGIN INSERT INTO @YeniStoklar (ID, ANABIRIM) 
SELECT i.ID, i.ANABIRIM FROM inserted i WHERE NOT EXISTS (SELECT 1 FROM dbo.STOKFIYAT sf WHERE sf.STOKID = i.ID); 
INSERT INTO dbo.STOKFIYAT (STOKID, FIYATADI, BIRIM, FIYAT, KUR, EKLEYEN, EKLEMETARIHI, DEGISTIREN, DEGISTIRMETARIHI, 
KDVDURUM, PAKETID, SATIS) SELECT ys.ID, g.DEGER, ys.ANABIRIM, -1, @Kur, NULL, GETDATE(), NULL, GETDATE(), 0, 0, 1 
FROM @YeniStoklar ys 
CROSS APPLY (SELECT TOP 1 DEGER FROM dbo.GENINI WHERE BOLUM = -1007 AND DIL = -1) g; 
INSERT INTO dbo.STOKFIYAT (STOKID, FIYATADI, BIRIM, FIYAT, KUR, EKLEYEN, EKLEMETARIHI, DEGISTIREN, 
DEGISTIRMETARIHI, KDVDURUM, PAKETID, SATIS) SELECT ys.ID, g.DEGER, ys.ANABIRIM, -1, @Kur, NULL, GETDATE(), NULL, 
GETDATE(), 0, 0, 0 FROM @YeniStoklar ys JOIN dbo.GENINI g ON g.BOLUM = -1008 AND g.DIL = -1; END; 
IF EXISTS (SELECT 1 FROM deleted) BEGIN DELETE sf FROM dbo.STOKFIYAT sf JOIN deleted d ON d.ID = sf.STOKID; 
END END

CREATE   TRIGGER [dbo].[TG_Uretim_StokFiyatGuncelle] on [dbo].[URETIMRECETEDETAY] after INSERT,DELETE,UPDATE
as
BEGIN
	SET NOCOUNT ON;

	IF EXISTS (SELECT 1 FROM inserted) AND EXISTS (SELECT 1 FROM deleted)
	BEGIN
		IF NOT (
			UPDATE(URUNID) OR
			UPDATE(ADET) OR
			UPDATE(ADETHESAP) OR
			UPDATE(KDVDURUM) OR
			UPDATE(URETIMRECETEID) OR
			UPDATE(ANAURUN) OR
			UPDATE(BIRIM)
		)
			RETURN;
	END;

	DECLARE  
		@insertedID int, @UrunID int, @FaturaTur int, @UretimID int, @UretimReceteID int, @AdetHesap int, @HizmetUrunTur int,
		@ToplamAdet Float, @AdetSon float, @PaketId FLOAT,
		@MaliyetOrt money, @MaliyetSon money,	
		@Tip VARCHAR(1), @SQL VARCHAR(8000),
		@Tarih DATETIME,
		@KdvDurum BIT;

    IF EXISTS ( SELECT * FROM deleted ) BEGIN
		SET @Tip = 'D'; 
		IF EXISTS ( SELECT * FROM INSERTED )
			SET @Tip = 'U';
	END ELSE BEGIN 
		SET @Tip = 'I';
	END;

	IF (@Tip IN ('I','U')) BEGIN
		select  @insertedID=ID,
				@UrunID=D.URUNID,
				@ToplamAdet=ADET,
				@MaliyetOrt=D.MALIYETORT,
				@MaliyetSon=MALIYETSON,
				@UretimID=(SELECT URUNID FROM URETIMRECETEDETAY URD WHERE URD.URETIMRECETEID=D.URETIMRECETEID AND ADET=1 AND ANAURUN=1),
				@Tarih=DEGISTIRMETARIHI,
				@UretimReceteID=URETIMRECETEID,
				@KdvDurum=kdvdurum
		from 
				INSERTED D;
		if ((@UretimID is not null) AND (@UretimID <> 0) AND (@ToplamAdet < 0)) BEGIN
			SET @AdetSon =(select ADETHESAP from URETIMRECETEDETAY U WHERE U.ID=@insertedID)/(select ADETHESAP from URETIMRECETEDETAY U WHERE U.URETIMRECETEID=@UretimReceteID AND ADETHESAP>0 AND ANAURUN=1);
			UPDATE URETIMRECETEDETAY SET 
				MALIYETSON=case when (select  FIYAT FROM STOKFIYAT SF WHERE FIYATADI=-1 AND URETIMRECETEDETAY.BIRIM=SF.BIRIM AND STOKID=@UrunID)>0.0 
								then (select  FIYAT FROM STOKFIYAT SF WHERE FIYATADI=-1 AND URETIMRECETEDETAY.BIRIM=SF.BIRIM AND STOKID=@UrunID) else 0.0 end,

				MALIYETORT=case when (select  FIYAT FROM STOKFIYAT SF WHERE FIYATADI=-2 AND URETIMRECETEDETAY.BIRIM=SF.BIRIM AND STOKID=@UrunID)>0.0 
								then (select  FIYAT FROM STOKFIYAT SF WHERE FIYATADI=-2 AND URETIMRECETEDETAY.BIRIM=SF.BIRIM AND STOKID=@UrunID)else 0.0 end,
				ADET =@AdetSon,
				MIKTAR =@AdetSon,
				DEGISTIRMETARIHI=@Tarih 
			WHERE ID=@insertedID;
		END;
		if (@kdvdurum=1) begin        
			UPDATE URETIMRECETEDETAY SET 
				MALIYETSON=(SELECT ABS(SUM(MALIYETSON*MIKTAR*(100.0+S.KDV)/100.0))  FROM URETIMRECETEDETAY URD INNER JOIN STOKLAR S ON S.ID= URD.URUNID WHERE URETIMRECETEID=@UretimReceteID AND ADET<0),
				MALIYETORT=(SELECT ABS(SUM(MALIYETORT*MIKTAR*(100.0+S.KDV)/100.0))  FROM URETIMRECETEDETAY URD INNER JOIN STOKLAR S ON S.ID= URD.URUNID WHERE URETIMRECETEID=@UretimReceteID AND ADET<0),
				DEGISTIRMETARIHI=@Tarih 
			WHERE URUNID=@UretimID AND ADET>0 AND ANAURUN=1;
		end else begin
			UPDATE URETIMRECETEDETAY SET 
				MALIYETSON=(SELECT SUM(MALIYETSON*MIKTAR) FROM URETIMRECETEDETAY URD WHERE URETIMRECETEID=@UretimReceteID AND ADET<0),
				MALIYETORT=(SELECT SUM(MALIYETORT*MIKTAR) FROM URETIMRECETEDETAY URD WHERE URETIMRECETEID=@UretimReceteID AND ADET<0),
				DEGISTIRMETARIHI=@Tarih 
			WHERE URUNID=@UretimID AND ADET>0 AND ANAURUN=1;
		end;      
    END;
END

----------8<---------- TRIGGER: TG_IMAJ_DosyaRefAzalt ----------8<----------
CREATE   TRIGGER dbo.TG_IMAJ_DosyaRefAzalt ON dbo.IMAJ AFTER DELETE
AS
BEGIN
  SET NOCOUNT ON;
  IF NOT EXISTS (SELECT 1 FROM deleted WHERE DOSYAID > 0) RETURN;
  UPDATE d SET REFSAYAC = d.REFSAYAC - x.c
    FROM dbo.DOSYA d
    JOIN (SELECT DOSYAID, COUNT(*) AS c FROM deleted WHERE DOSYAID > 0 GROUP BY DOSYAID) x
      ON x.DOSYAID = d.ID;
  DELETE d FROM dbo.DOSYA d
    JOIN (SELECT DISTINCT DOSYAID FROM deleted WHERE DOSYAID > 0) x ON x.DOSYAID = d.ID
    WHERE d.REFSAYAC <= 0;
END
