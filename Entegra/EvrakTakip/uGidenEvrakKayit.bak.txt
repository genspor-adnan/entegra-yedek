WITH Dizin AS
(
    SELECT ID,USTID,AD=CAST(AD AS NVARCHAR(260))
    FROM DOKUMANKLASOR
    WHERE USTID=0
    UNION ALL
    SELECT A.ID
    , A.USTID,
    AD=CAST(V.AD+'\'+A.AD AS NVARCHAR(260))
    FROM DOKUMANKLASOR A
    INNER JOIN Dizin V ON
        V.ID = A.USTID
)

select distinct
 D.*,
DTIP=1,KISAYOLID=0,Firma.FIRMA as KURUM,Lokasyon.ACIKLAMA as LOKASYONAD,
Sorumlu.FIRMA as SORUMLUAD,
EXT='.'+I.BELGETURU,I.SURUM,
I.BOYUT,
KLASORAD=(SELECT AD FROM Dizin where ID=D.KLASOR),
ONAYLAYACAKAD=(SELECT FIRMA FROM REHBER where ID=I.ONAYLAYACAK),
ONAYLAYANAD=(SELECT FIRMA FROM REHBER where ID=I.ONAY),
EKLEYENAD=(SELECT FIRMA FROM REHBER where ID=D.EKLEYEN),
DEGISTIRENAD=(SELECT FIRMA FROM REHBER where ID=D.DEGISTIREN)
from DOKUMAN D
 INNER JOIN IMAJ I ON I.ID = (select top 1 ID from IMAJ where YERI=1 AND YER_ID=D.ID order by ID desc)
 LEFT OUTER JOIN REHBER Firma on Firma.ID=D.REHBERID
 LEFT OUTER JOIN LOKASYON AS Lokasyon ON Lokasyon.ID=D.LOKASYON  
 LEFT OUTER JOIN REHBER AS Sorumlu ON Sorumlu.ID = I.REHBERID
 LEFT OUTER JOIN DOKUMANYETKI DY ON DY.YERI=321 AND DY.YERID = D.ID

select 
  distinct   D.*,DTIP=0,KISAYOLID=DK.ID,Firma.FIRMA as KURUM,Lokasyon.ACIKLAMA as LOKASYONAD,Sorumlu.FIRMA as 
SORUMLUAD,
EXT=case when D.AD like '%.%' then '.'+REVERSE( SUBSTRING(REVERSE(isnull(D.AD,'.')),1,CHARINDEX('.',REVERSE(isnull
(D.AD,'.')),1)-1)) else '' end,I.SURUM,
I.BOYUT,
KLASORAD=(SELECT AD FROM Dizin where ID=D.KLASOR),
ONAYLAYACAKAD=(SELECT FIRMA FROM REHBER where ID=I.ONAYLAYACAK),
ONAYLAYANAD=(SELECT FIRMA FROM REHBER where ID=I.ONAY),
EKLEYENAD=(SELECT FIRMA FROM REHBER where ID=D.EKLEYEN),
DEGISTIRENAD=(SELECT FIRMA FROM REHBER where ID=D.DEGISTIREN)
from 
	DOKUMANKISAYOL DK 
	inner join DOKUMAN D on DK.DOKUMANID=D.ID
	INNER JOIN IMAJ I ON I.ID = (select top 1 ID from IMAJ where YERI=1 AND YER_ID=D.ID order by ID desc)
 	LEFT OUTER JOIN REHBER Firma on Firma.ID=D.REHBERID
 	LEFT OUTER JOIN LOKASYON AS Lokasyon ON  Lokasyon.ID=D.LOKASYON  
 	LEFT OUTER JOIN REHBER AS Sorumlu ON Sorumlu.ID = I.REHBERID
